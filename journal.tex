% Macro to add a todo/task
\newcommand{\todo}[2][]{%
  % Increment the counter and make it referenceable
  \refstepcounter{todonotesnumber}%
  \phantomsection % Create an anchor here
  \def\todocolor{blue}%
    \TodoIsDonefalse%
  % Parse key=val options
  \setkeys{todo}{#1}%
	\ifTodoIsDone
    % Do not write to .tdo file
	  \addcontentsline{odo}{section}{~\thetodonotesnumber: \textcolor{\todocolor}{~ #2}}%
  \else
  % Add to Table of Contents
	  \addcontentsline{tdo}{section}{~\thetodonotesnumber:\textcolor{\todocolor}{~ #2}}%
\fi
  % Display in text
  \textbf{~\thetodonotesnumber:}~ #2\par
}
