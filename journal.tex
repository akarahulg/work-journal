\newcommand{\todotextformat}[1]{#1}

% --- MAIN TODO COMMAND ---
% This command now writes hyperlinked entries to the .tdo, .odo, .bgo files
\newcommand{\todo}[2][]{%
  \refstepcounter{todonotesnumber}\phantomsection%
  \hypertarget{todo-target-\thetodonotesnumber}{}% <--- Added hypertarget for links
  \@resetTodoState
  \setcounter{subtodocounter}{0}
  \setkeys{todo}{#1}%
  
  \ifTodoIsStruck\renewcommand{\todotextformat}[1]{\sout{##1}}\else\renewcommand{\todotextformat}[1]{##1}\fi
  \def\plantag{}
  
  \ifx\todoplandate\empty\else
      \def\plantag{\textcolor{TaskPlan}{\small~[Plan: \todoplandate]}}%
      \ifx\todocolor\unmodified\def\todocolor{TaskPlan}\fi
  \fi
  
  % Create the hyperlinked text for the list entry
  \protected@edef\listentrytext{\noexpand\hyperlink{todo-target-\thetodonotesnumber}{\thetodonotesnumber: #2}}
  
  \ifTodoIsDone
    \ifx\tododonedate\empty\edef\tododonedate{\zref@getcurrent{entrydate}}\fi
    \ifx\tododonedate\lastOdoDate\else
        \edef\myFormattedDate{\DTMlongdate{\tododonedate}}%
        \protected@edef\dateheadingtext{\noexpand\normalfont\noexpand\bfseries\noexpand\color{DateHeading}--- Completed on \myFormattedDate ---}%
        \addcontentsline{odo}{subsection}{\dateheadingtext}%
        \xdef\lastOdoDate{\tododonedate}%
    \fi
    \addcontentsline{odo}{section}{\listentrytext}%

    \def\taskicon{\boxtimes}\def\taskcolor{TaskDone}%
  \else
    \edef\creationdate{\zref@getcurrent{entrydate}}
    \ifx\creationdate\lastTdoDate\else
        \edef\myFormattedDate{\DTMlongdate{\creationdate}}%
        \protected@edef\dateheadingtext{\noexpand\normalfont\noexpand\bfseries\noexpand\color{DateHeading}--- Created on \myFormattedDate ---}%
        \addcontentsline{tdo}{subsection}{\dateheadingtext}%
        \xdef\lastTdoDate{\creationdate}%
    \fi
    \addcontentsline{tdo}{section}{\listentrytext}%

    \def\taskicon{\square}\def\taskcolor{\todocolor}%
  \fi

  \par\noindent\textbf{\thetodonotesnumber.}~{\Large\raisebox{-0.1em}{\textcolor{\taskcolor}{$\taskicon$}}}~\textcolor{\taskcolor}{\todotextformat{#2}}\plantag\par
}

% --- Backlog command with list date logic ---
\newcommand{\backlog}[2][]{%
    \refstepcounter{todonotesnumber}\phantomsection%
    \hypertarget{todo-target-\thetodonotesnumber}{}% <--- Added hypertarget for links
    \@resetTodoState \setkeys{todo}{color=Backlog, #1}%
    \setcounter{subtodocounter}{0}%
    \def\plantag{}
    
    \ifx\todoplandate\empty
        \edef\thedate{\zref@getcurrent{entrydate}}%
        \def\datetype{Created}%
    \else
        \edef\thedate{\todoplandate}%
        \def\plantag{\textcolor{TaskPlan}{\small~[Plan: \todoplandate]}}%
        \def\datetype{Planned for}%
    \fi

    % Create the hyperlinked text for the list entry
    \protected@edef\listentrytext{\noexpand\hyperlink{todo-target-\thetodonotesnumber}{\thetodonotesnumber: #2}}

    \ifx\thedate\lastBgoDate\else
        \edef\myFormattedDate{\DTMlongdate{\thedate}}%
        \protected@edef\dateheadingtext{\noexpand\normalfont\noexpand\bfseries\noexpand\color{DateHeading}--- \datetype\space on \myFormattedDate ---}%
        \addcontentsline{bgo}{subsection}{\dateheadingtext}%
        \xdef\lastBgoDate{\thedate}%
    \fi
    \addcontentsline{bgo}{section}{\listentrytext}%
    \par\noindent\textbf{\thetodonotesnumber.}~{\Large\raisebox{-0.1em}{\textcolor{\todocolor}{$\square$}}}~\textcolor{\todocolor}{#2}\plantag\par
}

% --- COMMANDS TO GENERATE LISTS (REWRITTEN) ---
\newcommand{\@listtasks}[1]{%
  \begingroup % Start a local group to keep redefinitions from leaking
    % Redefine how "subsection" level entries (our date headings) are formatted.
    % ##1 is the text, ##2 is the page number (which we ignore).
    % We add some vertical space to separate the date groups.
    \def\l@subsection##1##2{\par\addvspace{0.5em}\noindent##1\par\nopagebreak}%
    % Redefine how "section" level entries (our tasks) are formatted.
    % ##1 is the hyperlinked text, ##2 is the page number (which we ignore).
    \def\l@section##1##2{\par\noindent##1\par}%
    % Now, process the file with our custom formatting.
    \@starttoc{#1}%
  \endgroup % End the local group
}
