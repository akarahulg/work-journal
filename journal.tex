% --- Master Task Creation Commands (Internal) ---

% MODIFIED: Added \label to create a hyperlink anchor.
\renewcommand{\@createmaintask}[4][]{%
    \bgroup 
    \TaskIsDonefalse
    \def\taskcolor{#3}
    \setkeys{task}{#1} 
    \refstepcounter{tasknumber}
    \phantomsection
    \label{task:\thetasknumber} % <-- ADDED: Creates the target for the hyperlink
    \expandafter\gdef\csname tasktext@\thetasknumber\endcsname{#4}
    \ifTaskIsDone
        \addcontentsline{odo}{section}{\protect\numberline{\thetasknumber} #4}
    \else
        \addcontentsline{#2}{section}{\protect\numberline{\thetasknumber} \textcolor{#3}{#4}}
    \fi
    \par\noindent
    \ifTaskIsDone
        \textcolor{TaskDone}{\Large\raisebox{-0.1em}{$\checkmark$}~\textbf{\thetasknumber:} #4}%
    \else
        \textcolor{\taskcolor}{\Large\raisebox{-0.1em}{$\square$}~\textbf{\thetasknumber:} #4}%
    \fi
    \par\vspace{1mm}
    \egroup
}

% MODIFIED: Added \label to create a hyperlink anchor for subtasks.
\renewcommand{\@createsubtask}[3][]{%
    \bgroup
    \TaskIsDonefalse
    \def\taskcolor{SubTask}
    \setkeys{task}{#1}

    \refstepcounter{subtasknumber}
    \phantomsection
    \label{task:\thesubtasknumber} % <-- ADDED: Creates the target for the subtask hyperlink
    \expandafter\gdef\csname tasktext@\thesubtasknumber\endcsname{#3}

    \ifTaskIsDone
        \addcontentsline{odo}{subsection}{\protect\numberline{\thesubtasknumber} #3}
    \else
        \addcontentsline{#2}{subsection}{\protect\numberline{\thesubtasknumber} #3}
    \fi

    \par\noindent\hspace{2em}
    \ifTaskIsDone
        \textcolor{SubTaskDone}{\raisebox{-0.1em}{$\checkmark$}~\textbf{\thesubtasknumber:} #3}%
    \else
        \textcolor{\taskcolor}{\raisebox{-0.1em}{$\square$}~\textbf{\thesubtasknumber:} #3}%
    \fi
    \par\vspace{1mm}
    \egroup
}
